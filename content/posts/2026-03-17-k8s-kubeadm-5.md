### 1.5 安装配置chrony时间同步

{{< cmd role="all" title="集群全部节点配置时间同步" >}}
# 配置时区为 亚洲上海
timedatectl set-timezone Asia/Shanghai

apt update
apt install -y chrony

# 配置 chrony
cat <<EOF | sudo tee /etc/chrony/chrony.conf
# 包含额外配置文件的目录
confdir /etc/chrony/conf.d

# 公共NTP服务器，iburst加快初始同步
server ntp.aliyun.com iburst
server time.cloudflare.com iburst
server pool.ntp.org iburst

# DHCP动态获取的时间源目录
sourcedir /run/chrony-dhcp

# 额外时间源配置目录
sourcedir /etc/chrony/sources.d

# chrony密钥文件，用于NTP认证（可选）
keyfile /etc/chrony/chrony.keys

# drift文件，记录系统时钟漂移，用于长期纠正
driftfile /var/lib/chrony/chrony.drift

# ntp dump文件目录，保存时间服务快照
ntsdumpdir /var/lib/chrony

# 日志目录
logdir /var/log/chrony

# 最大允许偏差（秒），超过则触发警告或调整
maxupdateskew 100.0

# 同步本地硬件时钟到系统时钟（开机/关机时生效）
rtcsync

# 允许在启动或偏差过大时快速修正时间
makestep 1 3
EOF

# 启动并设置开机自启
systemctl enable --now chrony

# 验证同步状态
chronyc sources -v
chronyc tracking
{{< /cmd >}}

## 2. 安装并配置 CRI 容器运行时插件

Kubernetes集群要求使用符合容器运行时接口（CRI）的容器运行时，常见的有如下三个

- containerd
- CRI-O
- cri-dockerd(Docker 方案)
更多详见[容器运行时](https://kubernetes.io/zh-cn/docs/setup/production-environment/container-runtimes/#cri-versions)

### 2.1 安装 containerd

{{< cmd role="all" title="集群全部节点安装 containerd" >}}
sudo apt install -y ca-certificates curl gnupg
sudo mkdir -p /etc/apt/keyrings

curl -fsSL https://mirrors.aliyun.com/docker-ce/linux/ubuntu/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg

echo \
  "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://mirrors.aliyun.com/docker-ce/linux/ubuntu \
  $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

sudo apt update

sudo apt install -y containerd.io

# 查看containerd版本，本文使用containerd.io v2.2.2 版本
containerd --version
{{< /cmd >}}


### 2.2 配置 containerd

{{< cmd role="all" title="集群全部节点配置 containerd" >}}
# 生成默认配置
sudo mkdir -p /etc/containerd
containerd config default | sudo tee /etc/containerd/config.toml

# 配置pause 容器镜像源为阿里云镜像源
sudo sed -i 's#registry.k8s.io/pause:3.10#registry.aliyuncs.com/google_containers/pause:3.10#g' /etc/containerd/config.toml

# 配置 systemd cgroup 驱动
sudo sed -i 's/SystemdCgroup = false/SystemdCgroup = true/' /etc/containerd/config.toml

# 在[plugins."io.containerd.grpc.v1.cri".registry]行下
# 配置文件第54行的 config_path = "" 改为 config_path = "/etc/containerd/certs.d"
vim /etc/containerd/config.toml 

# 配置国内docker镜像加速
sudo mkdir -p /etc/containerd/certs.d/docker.io

cat <<EOF | sudo tee /etc/containerd/certs.d/docker.io/hosts.toml
[host."https://docker.gh-proxy.org"]
  capabilities = ["pull", "resolve"]

[host."https://docker.1ms.run"]
  capabilities = ["pull", "resolve"]

[host."https://docker.xuanyuan.me"]
  capabilities = ["pull", "resolve"]

# 阿里云镜像加速器
[host."https://registry.cn-hangzhou.aliyuncs.com"]
  capabilities = ["pull", "resolve"]

# 网易云镜像加速器
[host."https://hub-mirror.c.163.com"]
  capabilities = ["pull", "resolve"]

# 腾讯云镜像加速器
[host."https://mirror.ccs.tencentyun.com"]
  capabilities = ["pull", "resolve"]

# 中国科技大学镜像

  capabilities = ["pull", "resolve"]

# 重启 containerd
sudo systemctl daemon-reload
sudo systemctl enable --now containerd
sudo systemctl restart containerd
{{< /cmd >}}

### 2.3 安装配置 crictl

{{< cmd role="all" title="集群全部节点安装配置 crictl" >}}
# K8s集群与crictl的大版本号需要保持一致
VERSION="v1.34.0"

wget https://github.com/kubernetes-sigs/cri-tools/releases/download/$VERSION/crictl-$VERSION-linux-amd64.tar.gz
# 上面的链接下载异常，可尝试如下链接，走大陆github代理下载
wget https://gh-proxy.org/https://github.com/kubernetes-sigs/cri-tools/releases/download/$VERSION/crictl-$VERSION-linux-amd64.tar.gz

sudo tar zxvf crictl-$VERSION-linux-amd64.tar.gz -C /usr/local/bin
rm -f crictl-$VERSION-linux-amd64.tar.gz

# 配置 crictl 指向 containerd
cat <<EOF | sudo tee /etc/crictl.yaml
runtime-endpoint: unix:///run/containerd/containerd.sock
image-endpoint: unix:///run/containerd/containerd.sock
timeout: 10
debug: false
pull-image-on-create: false
EOF

# 检查是否安装成功并连上 containerd
crictl version

# 拉取busybox容器镜像测试
sudo crictl pull busybox:latest
{{< /cmd >}}

## 4. Control Plane节点初始化集群
kube-apiserver:v1.34.5
kube-controller-manager:v1.34.5
kube-scheduler:v1.34.5
kube-proxy:v1.34.5
coredns:v1.12.1
pause:3.10.1
etcd:3.6.5-0

{{< cmd role="prod-k8s-control-01" title="prod-k8s-control-01 节点初始化集群" >}}
# 拉取阿里云镜像
sudo kubeadm config images pull \
  --image-repository registry.aliyuncs.com/google_containers \
  --kubernetes-version=v1.34.5

# 初始化（Flannel 专用 CIDR）
sudo kubeadm init \
  --image-repository registry.aliyuncs.com/google_containers \
  --kubernetes-version=v1.34.5\
  --pod-network-cidr=10.244.0.0/16 \
  --service-cidr=10.96.0.0/12 \
  --cri-socket unix:///run/containerd/containerd.sock \
  --control-plane-endpoint "VIP:6443" \
  --upload-certs

# 配置 kubectl
mkdir -p $HOME/.kube
sudo cp -i /etc/kubernetes/admin.conf $HOME/.kube/config
sudo chown $(id -u):$(id -g) $HOME/.kube/config
{{< /cmd >}}

--control-plane-endpoint：统一入口（VIP / LB）
--upload-certs：用于后续节点加入时分发证书

## 5. 安装并配置 CNI 网络插件 Calico

Kubernetes 默认没有 Pod 网络，为了解决多节点组网核心问题，Kubernetes使用一个叫容器网络接口(CNI，Container Network Interface)的标准，因此在kubeadm init初始化之后必须安装一个 CNI 网络插件，常见的CNI插件如下：
- Flannel(推荐新手) 简单稳定，小集群常用，默认不支持 网络隔离(NetworkPolicy)
- Calico(生产常用) 支持 NetworkPolicy
- Cilium eBPF 高性能
- Canal

{{< cmd role="prod-k8s-control-01" title="prod-k8s-control-01 安装 CNI" >}}
wget https://raw.githubusercontent.com/projectcalico/calico/v3.26.0/manifests/calico.yaml

# 修改 CALICO_IPV4POOL_CIDR，默认为192.168.0.0/16
vim calico.yaml


kubectl apply -f calico.yaml

# 查看Calico相关容器状态是否均为Running
kubectl get pods -n calico-system
{{< /cmd >}}

## 节点维护

节点维护流程：cordon → drain → 维护 → uncordon

{{< /cmd >}}

{{< /cmd >}}
