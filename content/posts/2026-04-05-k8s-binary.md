## 3. CRI

Containerd有两种二进制安装包：

- 第一种是containerd-2.2.2-linux-amd64.tar.gz，仅包括containerd  containerd-shim-runc-v2  containerd-stress  ctr四个二进制文件，需要额外安装runc和 CNI plugins。
- 第二种是cri-containerd-cni-xxxx，包含runc和k8s里的所需要的相关文件。k8s集群里需要用到此包。虽然包含runC，但是依赖系统中的seccomp（安全计算模式，是一种限制容器调用系统资源的模式。）
- 生产环境推荐第一种，方便灵活调整。

{{< cmd role="prod-k8s-control-01" title="prod-k8s-control-01分发二进制组件与ca证书到worker节点" >}}
# ssh免密配置

# =========================
# 基础变量
# =========================
HOSTS=("prod-k8s-worker-01" "prod-k8s-worker-02")
IPS=("192.168.101.41" "192.168.101.42")

K8S_CA="${K8S_PKI_DIR}/ca.pem"
K8S_CA_KEY="${K8S_PKI_DIR}/ca-key.pem"

APISERVER="https://192.168.101.40:6443"

cd ${K8S_PKI_DIR} || exit 1

# =========================
# 循环处理每个 worker
# =========================
for i in ${!HOSTS[@]}; do
  NODE_NAME=${HOSTS[$i]}
  NODE_IP=${IPS[$i]}

  echo ">>> 处理节点: ${NODE_NAME} (${NODE_IP})"

  # =========================
  # 1. kubelet CSR
  # =========================
  cat <<EOF > kubelet-csr-${NODE_NAME}.json
{
  "CN": "system:node:${NODE_NAME}",
  "hosts": [
    "${NODE_NAME}",
    "${NODE_IP}"
  ],
  "key": {
    "algo": "rsa",
    "size": 4096
  },
  "names": [
    {
      "C": "CN",
      "L": "Beijing",
      "ST": "Beijing",
      "O": "system:nodes",
      "OU": "Kubernetes"
    }
  ]
}
EOF

  # =========================
  # 2. 签发 kubelet 证书
  # =========================
  cfssl gencert \
    -ca=${K8S_CA} \
    -ca-key=${K8S_CA_KEY} \
    -config=${K8S_PKI_DIR}/ca-config.json \
    -profile=kubernetes \
    kubelet-csr-${NODE_NAME}.json | cfssljson -bare kubelet-${NODE_NAME}

  # =========================
  # 3. kubelet kubeconfig
  # =========================
  kubectl config set-cluster kubernetes \
    --certificate-authority=${K8S_CA} \
    --embed-certs=true \
    --server=${APISERVER} \
    --kubeconfig=kubelet-${NODE_NAME}.kubeconfig

  kubectl config set-credentials system:node:${NODE_NAME} \
    --client-certificate=kubelet-${NODE_NAME}.pem \
    --client-key=kubelet-${NODE_NAME}-key.pem \
    --embed-certs=true \
    --kubeconfig=kubelet-${NODE_NAME}.kubeconfig

  kubectl config set-context default \
    --cluster=kubernetes \
    --user=system:node:${NODE_NAME} \
    --kubeconfig=kubelet-${NODE_NAME}.kubeconfig

  kubectl config use-context default \
    --kubeconfig=kubelet-${NODE_NAME}.kubeconfig

  # =========================
  # 4. kube-proxy 证书
  # =========================
  cat <<EOF > kube-proxy-csr-${NODE_NAME}.json
{
  "CN": "system:kube-proxy",
  "key": {
    "algo": "rsa",
    "size": 2048
  },
  "names": [
    {
      "C": "CN",
      "L": "Beijing",
      "ST": "Beijing",
      "O": "system:node-proxier",
      "OU": "Kubernetes"
    }
  ]
}
EOF

  cfssl gencert \
    -ca=${K8S_CA} \
    -ca-key=${K8S_CA_KEY} \
    -config=${K8S_PKI_DIR}/ca-config.json \
    -profile=kubernetes \
    kube-proxy-csr-${NODE_NAME}.json | cfssljson -bare kube-proxy-${NODE_NAME}

  # =========================
  # 5. kube-proxy kubeconfig
  # =========================
  kubectl config set-cluster kubernetes \
    --certificate-authority=${K8S_CA} \
    --embed-certs=true \
    --server=${APISERVER} \
    --kubeconfig=kube-proxy-${NODE_NAME}.kubeconfig

  kubectl config set-credentials system:kube-proxy \
    --client-certificate=kube-proxy-${NODE_NAME}.pem \
    --client-key=kube-proxy-${NODE_NAME}-key.pem \
    --embed-certs=true \
    --kubeconfig=kube-proxy-${NODE_NAME}.kubeconfig

  kubectl config set-context default \
    --cluster=kubernetes \
    --user=system:kube-proxy \
    --kubeconfig=kube-proxy-${NODE_NAME}.kubeconfig

  kubectl config use-context default \
    --kubeconfig=kube-proxy-${NODE_NAME}.kubeconfig

  # =========================
  # 6. 远程目录
  # =========================
  ssh root@${NODE_NAME} "mkdir -p /usr/local/bin /etc/kubernetes/pki /var/lib/kubelet /var/lib/kube-proxy"

  # =========================
  # 7. 分发二进制
  # =========================
  scp /usr/local/bin/{kubelet,kube-proxy} root@${NODE_NAME}:/usr/local/bin/

  # =========================
  # 8. 分发证书 + kubeconfig
  # =========================
  scp ${K8S_CA} root@${NODE_NAME}:/etc/kubernetes/pki/

  scp kubelet-${NODE_NAME}.pem root@${NODE_NAME}:/etc/kubernetes/pki/kubelet.pem
  scp kubelet-${NODE_NAME}-key.pem root@${NODE_NAME}:/etc/kubernetes/pki/kubelet-key.pem
  scp kubelet-${NODE_NAME}.kubeconfig root@${NODE_NAME}:${K8S_DIR}/kubelet.kubeconfig

  scp kube-proxy-${NODE_NAME}.pem root@${NODE_NAME}:/etc/kubernetes/pki/kube-proxy.pem
  scp kube-proxy-${NODE_NAME}-key.pem root@${NODE_NAME}:/etc/kubernetes/pki/kube-proxy-key.pem
  scp kube-proxy-${NODE_NAME}.kubeconfig root@${NODE_NAME}:${K8S_DIR}/kube-proxy.kubeconfig
done
