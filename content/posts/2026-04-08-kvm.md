```bash
sudo apt update

sudo apt install -y \
  qemu-kvm \
  libvirt-daemon-system \
  libvirt-clients \
  virtinst \
  bridge-utils \
  virt-manager

sudo systemctl enable --now libvirtd

virsh list --all

sudo usermod -aG libvirt $USER
newgrp libvirt
kvm-ok   # Ubuntu可能没有

virt-install \
  --name test-vm \
  --memory 2048 \
  --vcpus 2 \
  --disk size=20 \
  --cdrom /iso/ubuntu-24.04.iso \
  --network bridge=br0 \
  --os-variant ubuntu22.04 \
  --graphics none \
  --console pty,target_type=serial
```
