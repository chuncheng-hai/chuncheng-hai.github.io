### 2.2 部署配置HAproxy

大版本选择2.8 2028-Q2 (LTS)，LTS版本详见[HAProxy](https://www.haproxy.org/)
小版本选择2.8.20 last， last版本详见[HAProxy known bugs for maintenance branch 2.8](https://www.haproxy.org/bugs/bugs-2.8.html)

{{< cmd role="prod-k8s-lb-all" title="为全部LB服务器部署配置HAproxy" >}}
# 安装编译依赖
sudo apt update
sudo apt install -y \
  build-essential   \
  libssl-dev        \
  libpcre2-dev      \
  zlib1g-dev        \
  liblua5.4-dev     \
  libsystemd-dev    \
  libngtcp2-dev \
  libnghttp3-dev

# 下载 HAProxy 2.8.20 的源码包
wget https://www.haproxy.org/download/2.8/src/haproxy-2.8.20.tar.gz

# 解压源码包，并执行解压成功后清理
tar zxf haproxy-2.8.20.tar.gz && rm haproxy-2.8.20.tar.gz 

cd haproxy-2.8.20
make clean

make TARGET=linux-glibc \
     USE_OPENSSL=1      \
     USE_PCRE2=1 \
     USE_ZLIB=1  \
     USE_SYSTEMD=1 \
     USE_LUA=1    \
     USE_PROMEX=1 \
     -j$(nproc)

# 将 `${/app/haproxy}` 和 `${/app/haproxy/bin}` 替换为自定义的实际路径
make PREFIX=/usr/local       \
     SBINDIR=/usr/local/sbin \
     install 

haproxy -v

# 创建目录
mkdir -p /etc/haproxy /var/log/haproxy /run/haproxy

cat <<EOF | sudo tee /etc/haproxy/haproxy.cfg
# ========================
# 全局配置
# ========================
global
    log /dev/log local0        # 使用 syslog 记录日志（local0 facility）
    maxconn 100000            # HAProxy 最大并发连接数
    daemon                    # 以守护进程方式运行

# ========================
# 默认参数（frontend/backend继承）
# ========================
defaults
    log global                # 继承 global 中的日志配置
    mode tcp                  # 使用四层 TCP 模式
    option tcplog             # 使用 TCP 日志格式

    timeout connect 5s        # 后端连接超时
    timeout client  1m        # 客户端空闲超时
    timeout server  1m        # 后端响应超时

# ========================
# 前端（K8s API入口）
# ========================
frontend k8s_api
    bind *:6443                   # 监听K8s API Server 6443端口
    default_backend k8s_api_back  # 默认转发到后端集群

# ========================
# 后端（K8s control plane）
# ========================
backend k8s_api_back
    balance leastconn         # 最少连接负载均衡
    option tcp-check          # 启用 TCP 健康检查

    # fall 3：连续3次失败才判定下线
    # rise 2：连续2次成功恢复上线
    # maxconn：限制单节点最大连接数（防止压垮）

    server prod-k8s-control-01 10.142.100.65:6443 check fall 3 rise 2 maxconn 2000
    server prod-k8s-control-02 10.142.100.66:6443 check fall 3 rise 2 maxconn 2000
    server prod-k8s-control-03 10.142.100.67:6443 check fall 3 rise 2 maxconn 2000
EOF

# 验证配置
haproxy -c -f /etc/haproxy/haproxy.cfg

# 启动HAProxy
haproxy -f /etc/haproxy/haproxy.cfg
{{< /cmd >}}

{{< cmd role="prod-k8s-etcd-01" title="prod-k8s-etcd-01 配置证书" >}}
#  etcd 必须启用 TLS
mkdir -p /etc/etcd /var/lib/etcd
mkdir -p ~/etcd-cert && cd ~/etcd-cert

# 创建 CA
openssl genrsa -out ca.key 2048
openssl req -x509 -new -nodes -key ca.key -subj "/CN=etcd-ca" -days 10000 -out ca.crt

openssl genrsa -out etcd-01.key 2048

openssl req -new -key etcd-01.key \
  -subj "/CN=etcd-01" \
  -out etcd-01.csr

# 生成证书
openssl x509 -req -in etcd-01.csr \
  -CA ca.crt -CAkey ca.key -CAcreateserial \
  -out etcd-01.crt -days 365 \
  -extensions v3_req \
  -extfile <(cat <<EOF
[ v3_req ]
subjectAltName = IP:10.142.100.62
EOF
)

# 检查生成的文件，若没问题，拷贝至/etc/etcd/
cp -a ~/etcd-cert/*  /etc/etcd/

# systemd 配置
export ETCD_NAME=etcd-01
export ETCD_IP=10.142.100.62
export ETCD_CERT=etcd-01
# systemd 配置
cat <<EOF | sudo tee /etc/systemd/system/etcd.service
[Unit]
Description=etcd
After=network.target

[Service]
EnvironmentFile=/etc/etcd/etcd.env

ExecStart=/usr/local/bin/etcd \
  --name ${ETCD_NAME} \
  --data-dir /var/lib/etcd \
  --initial-advertise-peer-urls https://${ETCD_IP}:2380 \
  --listen-peer-urls https://${ETCD_IP}:2380 \
  --listen-client-urls https://${ETCD_IP}:2379 \
  --advertise-client-urls https://${ETCD_IP}:2379 \
  --initial-cluster etcd-01=https://10.142.100.62:2380,etcd-02=https://10.142.100.63:2380,etcd-03=https://10.142.100.64:2380 \
  --initial-cluster-state new \
  --initial-cluster-token etcd-cluster-1 \
  --cert-file=/etc/etcd/${ETCD_CERT}.crt \
  --key-file=/etc/etcd/${ETCD_CERT}.key \
  --trusted-ca-file=/etc/etcd/ca.crt \
  --peer-cert-file=/etc/etcd/${ETCD_CERT}.crt \
  --peer-key-file=/etc/etcd/${ETCD_CERT}.key \
  --peer-trusted-ca-file=/etc/etcd/ca.crt

Restart=always
RestartSec=5
LimitNOFILE=65536

[Install]
WantedBy=multi-user.target
EOF
{{< /cmd >}}

## 4. K8S各节点安装并配置 CRI 容器运行时插件

Kubernetes集群要求使用符合容器运行时接口（CRI）的容器运行时，常见的有如下三个

- containerd
- CRI-O
- cri-dockerd(Docker 方案)
更多详见[容器运行时](https://kubernetes.io/zh-cn/docs/setup/production-environment/container-runtimes/#cri-versions)

### 4.1 安装 containerd

{{< cmd role="all" title="集群全部节点安装 containerd" >}}
sudo apt install -y ca-certificates curl gnupg
sudo mkdir -p /etc/apt/keyrings

curl -fsSL https://mirrors.aliyun.com/docker-ce/linux/ubuntu/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg

echo \
  "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://mirrors.aliyun.com/docker-ce/linux/ubuntu \
  $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

sudo apt update

sudo apt install -y containerd.io

# 查看containerd版本，本文使用containerd.io v2.2.2 版本
containerd --version
{{< /cmd >}}


### 4.2 配置 containerd

{{< cmd role="all" title="集群全部节点配置 containerd" >}}

  capabilities = ["pull", "resolve"]

  capabilities = ["pull", "resolve"]

  capabilities = ["pull", "resolve"]

## 7. 安装并配置 CNI 网络插件

Kubernetes 默认没有 Pod 网络，为了解决多节点组网核心问题，Kubernetes使用一个叫容器网络接口(CNI，Container Network Interface)的标准，因此在kubeadm init初始化之后必须安装一个 CNI 网络插件，常见的CNI插件如下：
- Flannel(推荐新手) 简单稳定，小集群常用，默认不支持 网络隔离(NetworkPolicy)
- Calico(生产常用) 支持 NetworkPolicy
- Cilium eBPF 高性能
- Canal

{{< cmd role="prod-k8s-control-01" title="prod-k8s-control-01 安装 CNI" >}}
# 将flannel应用到集群，若网络异常，可使用github代理下载链接
kubectl apply -f  https://raw.githubusercontent.com/flannel-io/flannel/master/Documentation/kube-flannel.yml

# 下方为github代理下载链接，浏览器访问https://gh-proxy.com/可自行探索
kubectl apply -f  https://gh-proxy.org/https://raw.githubusercontent.com/flannel-io/flannel/master/Documentation/kube-flannel.yml


# 查看flannel相关的三个容器状态是否均为Running
kubectl get pod -A
{{< /cmd >}}
