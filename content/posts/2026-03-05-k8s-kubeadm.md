>本文部署环境：
>- 操作系统：Ubuntu 24.04 LTS
>- 硬件配置：2C4G，40G磁盘
>- 节点数量：3

- master
- node1
- node2


## 配置 hostname与hosts

{{< cmd role="master" target="k8s-master" title="设置主机名" >}}
hostnamectl set-hostname k8s-master
{{< /cmd >}}

{{< cmd role="node1" target="k8s-node1" title="设置主机名" >}}
hostnamectl set-hostname k8s-node1
{{< /cmd >}}

{{< cmd role="node2" target="k8s-node2" title="设置主机名" >}}
hostnamectl set-hostname k8s-node2
{{< /cmd >}}

{{< cmd role="all" target="master, node1, node2" title="基础系统配置（每台都执行）" >}}
192.168.1.10 k8s-master
192.168.1.11 k8s-node1
192.168.1.12 k8s-node2

# 永久关闭swap
sed -i '/swap/s/^/#/' /etc/fstab

# 加载内核模块
cat <<EOF | tee /etc/modules-load.d/k8s.conf
overlay
br_netfilter
EOF

modprobe overlay
modprobe br_netfilter

# 配置内核参数

cat <<EOF | tee /etc/sysctl.d/k8s.conf
net.bridge.bridge-nf-call-iptables=1
net.bridge.bridge-nf-call-ip6tables=1
net.ipv4.ip_forward=1
EOF

sysctl --system

{{< /cmd >}}


## 安装 containerd

{{< cmd role="all" target="master, node1, node2" title="安装并配置 containerd" >}}
apt update
apt install -y containerd

mkdir -p /etc/containerd
containerd config default > /etc/containerd/config.toml

# 将SystemdCgroup = false修改为 SystemdCgroup = true
vim /etc/containerd/config.toml

# 重启containerd
systemctl restart containerd
systemctl enable containerd
{{< /cmd >}}

{{< cmd role="all" target="master, node1, node2" title="安装 kubelet/kubeadm/kubectl" >}}
apt install -y apt-transport-https ca-certificates curl

cat <<EOF >/etc/apt/sources.list.d/kubernetes.list
deb https://mirrors.aliyun.com/kubernetes/apt/ kubernetes-xenial main
EOF

# 更新源
apt update

# 安装Kubernetes组件
apt install -y kubelet kubeadm kubectl

# 锁版本
apt-mark hold kubelet kubeadm kubectl

# 拉取 Kubernetes 阿里云镜像
kubeadm config images pull \
--image-repository registry.aliyuncs.com/google_containers

# 查看
ctr -n k8s.io images list
{{< /cmd >}}


{{< cmd role="master" target="k8s-master" title="初始化集群（仅主节点）" >}}
kubeadm init \
--apiserver-advertise-address=192.168.1.10 \
--image-repository registry.aliyuncs.com/google_containers \
--kubernetes-version v1.29.0 \
--service-cidr=10.96.0.0/12 \
--pod-network-cidr=192.168.0.0/16

mkdir -p $HOME/.kube

cp -i /etc/kubernetes/admin.conf $HOME/.kube/config

chown $(id -u):$(id -g) $HOME/.kube/config

curl https://docs.projectcalico.org/manifests/calico.yaml -O

kubectl apply -f calico.yaml

kubectl get pods -n kube-system
{{< /cmd >}}

部署 nginx
kubectl create deployment nginx --image=nginx
暴露服务
kubectl expose deployment nginx --port=80 --type=NodePort

安装Helm
curl https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3 | bash
验证
helm version
