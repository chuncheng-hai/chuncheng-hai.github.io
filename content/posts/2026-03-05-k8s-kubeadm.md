>本文部署环境：
>- 操作系统：Ubuntu 24.04 LTS
>- 硬件配置：2C4G，40G磁盘
>- 节点数量：3

节点命名规则：
pro：生产环境
k8s：kubernetes
control节点 master
worker节点

- prod-k8s-control-01 192.168.101.40 
- prod-k8s-worker-01  192.168.101.41
- prod-k8s-worker-02  192.168.101.42

## 1. 安装和配置先决条件

### 1.1 配置 hostname 主机名

{{< cmd role="prod-k8s-control-01" title="prod-k8s-control-01 配置主机名" >}}
sudo hostnamectl set-hostname prod-k8s-control-01
{{< /cmd >}}

{{< cmd role="prod-k8s-worker-01" title="prod-k8s-worker-01 配置主机名" >}}
sudo hostnamectl set-hostname prod-k8s-worker-01
{{< /cmd >}}

{{< cmd role="prod-k8s-worker-02" title="prod-k8s-worker-02 配置主机名" >}}
sudo hostnamectl set-hostname prod-k8s-worker-02
{{< /cmd >}}

### 1.2 配置 /etc/hosts 映射

{{< cmd role="all" title="集群全部节点配置IP与主机名的映射" >}}
# 备份/etc/hosts 文件
sudo cp /etc/hosts \
/etc/hosts.$(date +%F_%H%M%S).before-k8s-cluster-install.bak

cat <<EOF | sudo tee /etc/hosts
192.168.101.40  prod-k8s-control-01
192.168.101.41  prod-k8s-worker-01
192.168.101.42  prod-k8s-worker-02
EOF

{{< /cmd >}}



### 1.3 切换阿里云 apt 源

{{< cmd role="all" title="集群全部节点配置切换阿里云 apt 源" >}}
bash <(curl -sSL https://linuxmirrors.cn/main.sh) \
  --source mirrors.aliyun.com \
  --protocol https            \
  --use-intranet-source false \
  --backup true               \
  --upgrade-software false    \
  --clean-cache false         \
  --ignore-backup-tips        \
  --pure-mode

sudo apt update && sudo apt upgrade -y
{{< /cmd >}}

### 1.4 关闭 swap + 防火墙

{{< cmd role="all" title="集群全部节点配置关闭 swap + 防火墙" >}}
sudo swapoff -a
sudo sed -i '/swap/s/^/#/' /etc/fstab
sudo ufw disable
{{< /cmd >}}

### 1.5 加载内核模块

{{< cmd role="all" title="集群全部节点配置加载内核模块" >}}
cat <<EOF | sudo tee /etc/modules-load.d/k8s.conf
overlay
br_netfilter
EOF

sudo modprobe overlay
sudo modprobe br_netfilter

# 启用 IPv4 数据包转发
cat <<EOF | sudo tee /etc/sysctl.d/k8s.conf
net.bridge.bridge-nf-call-iptables  = 1
net.bridge.bridge-nf-call-ip6tables = 1
net.ipv4.ip_forward                 = 1
EOF

# 应用 sysctl 参数而不重新启动
sudo sysctl --system
{{< /cmd >}}


## 2. 安装并配置容器运行时

Kubernetes集群要求使用符合容器运行时接口（CRI）的容器运行时

### 2.1 安装 containerd

{{< cmd role="all" title="集群全部节点安装 containerd" >}}
sudo apt install -y ca-certificates curl gnupg
sudo mkdir -p /etc/apt/keyrings

curl -fsSL https://mirrors.aliyun.com/docker-ce/linux/ubuntu/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg

echo \
  "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://mirrors.aliyun.com/docker-ce/linux/ubuntu \
  $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

sudo apt update

sudo apt install -y containerd.io

# 查看containerd版本，本文使用containerd.io v2.2.2 版本
containerd --version
{{< /cmd >}}


### 2.2 配置 containerd

{{< cmd role="all" title="集群全部节点配置 containerd" >}}
# 生成默认配置
sudo mkdir -p /etc/containerd
containerd config default | sudo tee /etc/containerd/config.toml

sudo sed -i 's#registry.k8s.io/pause:3.10#registry.aliyuncs.com/google_containers/pause:3.10#g' /etc/containerd/config.toml
# 配置 systemd cgroup 驱动
sudo sed -i '/SystemdCgroup = false/a \            SystemdCgroup = true' /etc/containerd/config.toml

# 添加国内docker镜像加速
cat <<EOF | sudo tee -a /etc/containerd/config.toml

[plugins."io.containerd.grpc.v1.cri".registry.mirrors."docker.io"]
  endpoint = ["https://docker.m.daocloud.io", "https://registry.docker-cn.com"]

[plugins."io.containerd.grpc.v1.cri".registry.mirrors."quay.io"]
  endpoint = ["https://quay.m.daocloud.io"]
EOF

# 重启 containerd
sudo systemctl daemon-reload
sudo systemctl enable --now containerd
sudo systemctl restart containerd
{{< /cmd >}}

## 3. 安装kubeadm kubelet  kubectl

{{< cmd role="all" title="集群全部节点安装kubeadm kubelet  kubectl" >}}
# 添加阿里云 kubernetes apt 源
sudo apt update && sudo apt install -y apt-transport-https ca-certificates curl gnupg

curl -fsSL https://mirrors.aliyun.com/kubernetes/apt/doc/apt-key.gpg | sudo gpg --dearmor -o /etc/apt/keyrings/kubernetes-aliyun.gpg

echo "deb [signed-by=/etc/apt/keyrings/kubernetes-aliyun.gpg] https://mirrors.aliyun.com/kubernetes/apt/ kubernetes-xenial main" | sudo tee /etc/apt/sources.list.d/kubernetes.list

sudo apt update

# 安装指定版本
sudo apt install -y kubelet=1.34.5-1.1 kubeadm=1.34.5-1.1 kubectl=1.34.5-1.1

# 锁定版本，防止自动升级
sudo apt-mark hold kubelet kubeadm kubectl

# 配置kubelet开机自启并启动
sudo systemctl enable --now kubelet
{{< /cmd >}}

## 4. Control 节点初始化集群

{{< cmd role="prod-k8s-control-01" title="prod-k8s-control-01 节点初始化集群" >}}
# 拉取阿里云镜像
sudo kubeadm config images pull \
  --image-repository registry.aliyuncs.com/google_containers \
  --kubernetes-version=v1.34.5

# 初始化（Flannel 专用 CIDR）
sudo kubeadm init \
  --image-repository registry.aliyuncs.com/google_containers \
  --kubernetes-version=v1.34.5\
  --pod-network-cidr=10.244.0.0/16 \
  --service-cidr=10.96.0.0/12 \
  --cri-socket unix:///run/containerd/containerd.sock

# 配置 kubectl
mkdir -p $HOME/.kube
sudo cp -i /etc/kubernetes/admin.conf $HOME/.kube/config
sudo chown $(id -u):$(id -g) $HOME/.kube/config
{{< /cmd >}}

## 5. 安装 CNI

解决多节点组网核心问题，Kubernetes使用一个叫容器网络接口(CNI，Container Network Interface)的标准。

{{< cmd role="prod-k8s-control-01" title="prod-k8s-control-01 安装 CNI" >}}
wget https://raw.githubusercontent.com/flannel-io/flannel/master/Documentation/kube-flannel.yml

# 替换主 flannel 镜像（v0.28.1） - 出现在 initContainers 和 containers 中
sed -i 's|ghcr.io/flannel-io/flannel:v0.28.1|registry.cn-hangzhou.aliyuncs.com/flannel-io/flannel:v0.28.1|g' kube-flannel.yml

# 替换 flannel-cni-plugin 镜像（v1.9.0-flannel1）
sed -i 's|ghcr.io/flannel-io/flannel-cni-plugin:v1.9.0-flannel1|registry.cn-hangzhou.aliyuncs.com/flannel-io/flannel-cni-plugin:v1.9.0-flannel1|g' kube-flannel.yml

# dry-run 检查配置文件语法
kubectl apply -f kube-flannel.yml --dry-run=client

# 应用到集群
kubectl apply -f kube-flannel.yml
{{< /cmd >}}
