>部署注意： 
>- 整个 Elastic Stack 必须使用同一个版本号(本文选择8.19.12稳定版本)。因为Elasticsearch、Logstash、Kibana与Filebeat都是 Elastic Stack 产品，所以在部署过程中，各组件必须基于相同的版本号安装。
> - 部署顺序：Elasticsearch -> Logstash -> Kibana -> Filebeat
>- 部署全过程基于elastic用户进行操作

```bash
# 清理锁
rm -f /var/lib/dpkg/lock-frontend
rm -f /var/lib/apt/lists/lock
dpkg --configure -a
# 关闭Ubuntu 24.04 自动更新，避免影响系统
systemctl disable --now  unattended-upgrades
systemctl disable --now apt-daily.timer
systemctl disable --now apt-daily-upgrade.timer

# 删掉全部已安装的 Snap 软件
# 先删所有非 core / snapd
for p in $(snap list --all | awk 'NR>1 {print $1}' | grep -vE 'core|snapd'); do
  snap remove --purge $p
done
# 删 core
snap remove --purge core20
snap remove --purge core18
# 删 snapd
snap remove --purge snapd
sudo systemctl stop snapd
sudo systemctl disable --now snapd.socket

sudo apt purge -y snapd
sudo apt autoremove --purge -y
rm -rf ~/snap
rm -rf /snap
rm -rf /var/snap
rm -rf /var/lib/snapd
rm -rf /var/cache/snapd

cat <<EOF | sudo tee /etc/apt/preferences.d/no-snap.pref
Package: snapd
Pin: release a=*
Pin-Priority: -10
EOF

systemctl disable --now cloud-init cloud-init-local cloud-config cloud-final

# 移除无用软件包
apt purge -y \
    popularity-contest \
    ubuntu-advantage-tools \
    landscape-common \
    apport

apt autoremove --purge -y

cat >> /etc/sysctl.conf <<EOF

# 减少 swap 使用（云主机建议）
vm.swappiness = 10

# 减少 inode/dentry cache 抖动
vm.vfs_cache_pressure = 50

# 提高网络性能
net.core.somaxconn = 1024
net.core.netdev_max_backlog = 5000

# TCP 优化
net.ipv4.tcp_tw_reuse = 1
net.ipv4.tcp_fin_timeout = 15

# 增加本地临时端口范围，减少 TCP三次握手的TIME_WAIT
net.ipv4.ip_local_port_range = 1024 65535

EOF

sysctl -p

# 关闭 Transparent Huge Page（THP）透明大页
echo never > /sys/kernel/mm/transparent_hugepage/enabled'
echo 'echo never > /sys/kernel/mm/transparent_hugepage/enabled' >> /etc/rc.local

```

## 部署Elasticsearch

本节内容基于[Elasticsearch官方部署文档](https://www.elastic.co/guide/en/elasticsearch/reference/8.12/install-elasticsearch.html)


```bash
apt  install -y aria2
aria2c -x 16 -s 16 -k 1M https://artifacts.elastic.co/downloads/elasticsearch/elasticsearch-8.19.12-linux-x86_64.tar.gz

tar -xf elasticsearch-8.19.12-linux-x86_64.tar.gz -C /opt
mv /opt/elasticsearch-8.19.12 /opt/elasticsearch

# 创建  elasticsearch 用户
useradd -r -s /sbin/nologin elasticsearch
chown -R elasticsearch:elasticsearch /opt/elasticsearch
# 创建数据目录、日志目录
mkdir -p /data/elasticsearch /log/elasticsearch
chown -R elasticsearch:elasticsearch /data/elasticsearch  /log/elasticsearch


sudo tee /etc/sysctl.d/elasticsearch.conf <<EOF
# Increase virtual memory map count
vm.max_map_count=262144
# Increase maximum file descriptors system-wide
fs.file-max=65536
EOF

sysctl -p /etc/sysctl.d/elasticsearch.conf

cat <<EOF > /etc/security/limits.d/elasticsearch.conf
elasticsearch soft nofile 65536
elasticsearch hard nofile 65536
elasticsearch soft memlock unlimited
elasticsearch hard memlock unlimited
EOF

# 配置JVM Heap
cat <<EOF > /opt/elasticsearch/config/jvm.options
-Xms1g
-Xmx1g
EOF

cat <<EOF | sudo tee /opt/elasticsearch/config/elasticsearch.yml
cluster.name: prod-es-cluster
node.name: node-1

node.roles: [master, data, ingest]

path.data: /data/elasticsearch
path.logs: /log/elasticsearch

network.host: 0.0.0.0
http.port: 9200

bootstrap.memory_lock: true

discovery.seed_hosts:
  - 192.168.101.40
  - 192.168.101.44
  - 192.168.101.45

cluster.initial_master_nodes:
  - prod-es-01
  - prod-es-02
  - prod-es-03

# ===== TLS 核心 =====
xpack.security.enabled: true

xpack.security.transport.ssl.enabled: true
xpack.security.transport.ssl.verification_mode: certificate

xpack.security.transport.ssl.key: /opt/elasticsearch/node-1/node-1.key
xpack.security.transport.ssl.certificate: /opt/elasticsearch/node-1/node-1.crt
xpack.security.transport.ssl.certificate_authorities:
  - /opt/elasticsearch/ca/ca.crt

# （可选）HTTP TLS
xpack.security.http.ssl.enabled: false
EOF

cat <<EOF > /etc/systemd/system/elasticsearch.service
[Unit]
Description=Elasticsearch
After=network.target

[Service]
Type=simple
User=elasticsearch
Group=elasticsearch

# JAVA_HOME Env
Environment=ES_HOME=/opt/elasticsearch
Environment=ES_JAVA_HOME=/opt/elasticsearch/jdk

ExecStart=/opt/elasticsearch/bin/elasticsearch
Restart=always

LimitNOFILE=65536
LimitNPROC=4096
LimitMEMLOCK=infinity

# allow memory lock
AmbientCapabilities=CAP_IPC_LOCK

[Install]
WantedBy=multi-user.target
EOF

systemctl daemon-reexec
systemctl daemon-reload
systemctl enable --now elasticsearch
systemctl status elasticsearch

# 验证最大打开文件句柄数是否为65536
cat /proc/$(systemctl show -p MainPID --value elasticsearch)/limits | grep "Max open files"


```


```bash
cd /opt/elasticsearch/

# CA
/opt/elasticsearch/bin/elasticsearch-certutil ca --pem --silent --out ca.zip

apt install -y unzip
ls ca.zip && unzip ca.zip

cat <<EOF | sudo tee instances.yml
instances:
  - name: prod-es-01
    ip: [192.168.101.40]
  - name: prod-es-02
    ip: [192.168.101.44]
  - name: prod-es-03
    ip: [192.168.101.45]
EOF

# cert
/opt/elasticsearch/bin/elasticsearch-certutil cert \
  --pem \
  --in instances.yml \
  --ca-cert ca/ca.crt \
  --ca-key ca/ca.key \
  --silent \
  --out certs.zip

ls certs.zip && unzip certs.zip

for host in prod-es-02 prod-es-03; do
  scp -r /opt/elasticsearch/ca root@$host:/opt/elasticsearch/
  scp -r /opt/elasticsearch/$host root@$host:/opt/elasticsearch/
done

chown -R elasticsearch:elasticsearch /opt/elasticsearch/node-*
chown -R elasticsearch:elasticsearch /opt/elasticsearch/ca
chmod 640 /opt/elasticsearch/node-*/*
```

## 部署Logstash

[Logstash官方部署文档](https://www.elastic.co/docs/reference/logstash/installing-logstash)
[Logstash各版本下载地址](https://www.elastic.co/downloads/logstash)

```bash
wget https://artifacts.elastic.co/downloads/logstash/logstash-8.19.12-linux-x86_64.tar.gz

tar -xf logstash-8.19.12-linux-x86_64.tar.gz -C /opt
mv /opt/logstash-8.19.12 /opt/logstash

useradd -r -s /sbin/nologin logstash
chown -R logstash:logstash /opt/logstash

mkdir -p /data/logstash /log/logstash
chown -R logstash:logstash /data/logstash /log/logstash

cat <<EOF > /opt/logstash/config/logstash.conf
input {
  stdin { }
}

output {
  stdout { codec => rubydebug }
}
EOF

cat <<EOF >> /opt/logstash/config/logstash.yml
path.data: /data/logstash
path.logs: /log/logstash
EOF

cat <<EOF > /etc/systemd/system/logstash.service
[Unit]
Description=Logstash
After=network.target

[Service]
Type=simple
User=logstash
Group=logstash

Environment=LS_HOME=/opt/logstash

ExecStart=/opt/logstash/bin/logstash -f /opt/logstash/config/logstash.conf

Restart=always
LimitNOFILE=65536

[Install]
WantedBy=multi-user.target
EOF

systemctl daemon-reload
systemctl enable --now logstash
```



编辑/etc/logstash/conf.d/logstash.conf配置文件
pipeline

```
input {
  beats {
    port => 5044
  }
}

output {
  elasticsearch {
    hosts => ["localhost:9200"]
  }
}
```

sudo /bin/systemctl daemon-reload
sudo /bin/systemctl --now logstash.service

## 部署Filebeat

[Filebeat官方部署文档](https://www.elastic.co/docs/reference/beats/filebeat/filebeat-installation-configuration)

curl -L -O https://artifacts.elastic.co/downloads/beats/filebeat/filebeat-8.19.12-amd64.deb
sudo dpkg -i filebeat-8.19.12-amd64.deb

curl -L -O https://artifacts.elastic.co/downloads/beats/filebeat/filebeat-8.19.12-x86_64.rpm
sudo rpm -vi filebeat-8.19.12-x86_64.rpm

```bash
curl -L -O https://artifacts.elastic.co/downloads/beats/filebeat/filebeat-8.19.12-linux-x86_64.tar.gz

tar xzvf filebeat-8.19.12-linux-x86_64.tar.gz

./filebeat modules enable nginx
```

/etc/filebeat/filebeat.yml

filebeat.inputs:
- type: log
  paths:
    - /var/log/*.log

output.logstash:
  hosts: ["localhost:5044"]

sudo /bin/systemctl daemon-reload
sudo /bin/systemctl enable --now filebeat.service


日志收集数据流：应用服务器(日志产生者) → Filebeat(采集代理) → Logstash(日志处理引擎 ) → Elasticsearch(日志存储与搜索引擎) → Kibana(日志可视化平台)
