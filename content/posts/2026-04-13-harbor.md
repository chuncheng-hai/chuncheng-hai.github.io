## 部署Docker & Compose

```bash
apt remove $(dpkg --get-selections docker.io docker-compose docker-doc podman-docker containerd runc | cut -f1)

apt-get update
apt-get install ca-certificates curl

install -m 0755 -d /etc/apt/keyrings
curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
chmod a+r /etc/apt/keyrings/docker.asc
tee /etc/apt/sources.list.d/docker.sources <<EOF
Types: deb
URIs: https://mirrors.tuna.tsinghua.edu.cn/docker-ce/linux/ubuntu
Suites: $(. /etc/os-release && echo "$VERSION_CODENAME")
Components: stable
Signed-By: /etc/apt/keyrings/docker.asc
EOF

apt-get update
apt-get install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

docker version
docker compose version
```

## 配置systemd timer定期清理Harbor GC

```bash
# 查询GC状态
curl -u admin:password \
"https://harbor.example.com/api/v2.0/system/gc"

cat > /usr/local/bin/harbor-gc.sh <<'EOF'
#!/bin/bash

HARBOR="https://harbor.example.com"
USER="admin"
PASS="password"

curl -sk -u "$USER:$PASS" -X POST "$HARBOR/api/v2.0/system/gc"
EOF

chmod +x /usr/local/bin/harbor-gc.sh

cat > /etc/systemd/system/harbor-gc.service <<EOF
[Unit]
Description=Harbor GC

[Service]
Type=oneshot
ExecStart=/usr/local/bin/harbor-gc.sh
EOF

cat > /etc/systemd/system/harbor-gc.timer <<EOF
[Unit]
Description=Run Harbor GC weekly

[Timer]
OnCalendar=Sun 02:00
Persistent=true

[Install]
WantedBy=timers.target
EOF

systemctl daemon-reload
systemctl enable --now harbor-gc.timer
```
