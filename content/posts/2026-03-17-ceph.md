```bash
# 更新系统
apt update && apt upgrade -y

# 安装必备软件
apt install -y chrony lvm2 podman openssh-server

# 时间同步（企业必备）
systemctl enable --now chrony
# 配置 NTP 服务器（示例：阿里云或内部）
echo "server ntp.aliyun.com iburst" >> /etc/chrony/chrony.conf
systemctl restart chrony

# 主机名与 hosts（所有节点一致）
hostnamectl set-hostname node1   # 分别改 node2/node3
cat >> /etc/hosts << EOF
192.168.10.11 node1
192.168.10.12 node2
192.168.10.13 node3
EOF

# 防火墙（ufw，企业推荐限制来源）
ufw allow 22/tcp
ufw allow from 192.168.10.0/24 to any port 6789 proto tcp
ufw allow from 192.168.10.0/24 to any port 6800:7568 proto tcp
# 如果有独立集群网络，再加：
# ufw allow from 192.168.20.0/24 to any port 6800:7568 proto tcp
ufw --force enable
ufw status
```
