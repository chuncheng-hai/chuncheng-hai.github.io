本文目标为学习Ceph集群，所以节点配置较低
3 MON + 1 MGR + 3 OSD
dev-ceph-01 bootstrap + mon + mgr + osd 192.168.101.40 Ubuntu 24.04 LTS 2C4G  SSD 100G 
dev-ceph-02 mon + osd 192.168.101.44  Ubuntu 24.04 LTS 2C4G  SSD 100G 
dev-ceph-03 mon + osd 192.168.101.45  Ubuntu 24.04 LTS 2C4G  SSD 100G 

```bash
# 配置主机名
hostnamectl set-hostname dev-ceph-01
hostnamectl set-hostname dev-ceph-02
hostnamectl set-hostname dev-ceph-03

# 配置 hosts
cat >> /etc/hosts <<EOF
192.168.101.40 dev-ceph-01
192.168.101.44 dev-ceph-02
192.168.101.45 dev-ceph-03
EOF

# 清理锁
rm -f /var/lib/dpkg/lock-frontend
rm -f /var/lib/apt/lists/lock
dpkg --configure -a
# 关闭Ubuntu 24.04 自动更新，避免影响系统
systemctl disable --now  unattended-upgrades
systemctl disable --now apt-daily.timer
systemctl disable --now apt-daily-upgrade.timer

# 关闭 swap
swapoff -a
sed -i '/swap/d' /etc/fstab

bash <(curl -sSL https://linuxmirrors.cn/main.sh) \
  --source mirrors.aliyun.com \
  --protocol https            \
  --use-intranet-source false \
  --backup true               \
  --upgrade-software false    \
  --clean-cache false         \
  --ignore-backup-tips        \
  --pure-mode

# 配置时区为 亚洲上海
timedatectl set-timezone Asia/Shanghai

apt update
apt install -y chrony

# 配置 chrony
cat <<EOF | sudo tee /etc/chrony/chrony.conf
# 包含额外配置文件的目录
confdir /etc/chrony/conf.d

# 公共NTP服务器，iburst加快初始同步
server ntp.aliyun.com iburst
server time.cloudflare.com iburst
server pool.ntp.org iburst

# DHCP动态获取的时间源目录
sourcedir /run/chrony-dhcp

# 额外时间源配置目录
sourcedir /etc/chrony/sources.d

# chrony密钥文件，用于NTP认证（可选）
keyfile /etc/chrony/chrony.keys

# drift文件，记录系统时钟漂移，用于长期纠正
driftfile /var/lib/chrony/chrony.drift

# ntp dump文件目录，保存时间服务快照
ntsdumpdir /var/lib/chrony

# 日志目录
logdir /var/log/chrony

# 最大允许偏差（秒），超过则触发警告或调整
maxupdateskew 100.0

# 同步本地硬件时钟到系统时钟（开机/关机时生效）
rtcsync

# 允许在启动或偏差过大时快速修正时间
makestep 1 3
EOF

# 启动并设置开机自启
systemctl enable --now chrony

# 禁止ipv6解析
cat >> /etc/sysctl.conf <<EOF
net.ipv6.conf.all.disable_ipv6 = 1
net.ipv6.conf.default.disable_ipv6 = 1
net.ipv6.conf.lo.disable_ipv6 = 1
EOF

sysctl -p
```

## 安装docker

cephadm bootstrap 初始化集群需要docker(只安装containerd bootstrap 初始化不识别容器引擎)或podman
```bash
# 安装docker
apt update
apt install -y docker.io
mkdir -p /etc/docker

# 配置中国大陆加速镜像源
cat > /etc/docker/daemon.json <<EOF
{
  "registry-mirrors": [
    "https://registry.cn-hangzhou.aliyuncs.com",
    "https://docker.xuanyuan.me",
    "https://docker.gh-proxy.org",
    "https://hub-mirror.c.163.com",
    "https://mirror.ccs.tencentyun.com",
    "https://docker.mirrors.ustc.edu.cn"
  ]
}
EOF

systemctl daemon-reexec
systemctl restart docker
```

## 安装 Ceph

只在 dev-ceph-01 执行
```bash
# 安装 cephadm，版本号对应代码 pacific = Ceph 16.x quincy = Ceph 17.x reef = Ceph 18.x
wget -4 https://download.ceph.com/rpm-reef/el9/noarch/cephadm
# 校验文件类型 cephadm: a /usr/bin/python3 -s script executable (Zip archive)
file cephadm
chmod +x cephadm
mv cephadm /usr/sbin/
cephadm version

# 此处基于docker的/etc/docker/daemon.json文件配置的镜像源拉取镜像
cephadm bootstrap --mon-ip 192.168.101.40 --cluster-network 192.168.101.0/24
# 基于输出的URL(将其中的hostname替换为ip)、用户名、密码登陆，修改密码，重新登陆

# 因为是容器部署，所以ceph命令在容器中，需要配置cephadm shell --
alias ceph="cephadm shell -- ceph"
ceph -s

# 添加其它节点进入集群，添加原理是基于/etc/ceph/ceph.pub SSH连接节点自动部署组件，故需先配置SSH免密
ssh-copy-id -f -i /etc/ceph/ceph.pub root@192.168.101.44
ceph orch host add dev-ceph-02 192.168.101.44

ssh-copy-id -f -i /etc/ceph/ceph.pub root@192.168.101.45
ceph orch host add dev-ceph-03 192.168.101.45

# 验证其它节点是否加入集群，应该可以看到3 hosts in cluster
ceph orch host ls

# 部署 MON（形成 quorum）
ceph orch apply mon --placement="3 dev-ceph-01 dev-ceph-02 dev-ceph-03"
# 查看 MON 实际状态，若配置较低，需要等待十几分钟，3台节点全部running
ceph orch ps --daemon-type mon

# 部署 MGR
ceph orch apply mgr --placement="2 dev-ceph-01 dev-ceph-02"
ceph orch ps --daemon-type mgr

# 指定磁盘
ceph orch device ls
ceph orch daemon add osd dev-ceph-01:/dev/sdb
ceph orch daemon add osd dev-ceph-02:/dev/sdb
ceph orch daemon add osd dev-ceph-03:/dev/sdb
# 查看 OSD 状态
ceph osd tree

# 创建存储池
ceph osd pool create test-pool 32
# 设置副本数
ceph osd pool set test-pool size 3

rados put test1 /etc/hosts -p test-pool
rados get test1 /tmp/test1 -p test-pool

# 维护节点
ceph orch host maintenance enter dev-ceph-02
ceph orch host maintenance exit dev-ceph-02

# 维护整个ceph集群
# 标记 noout
ceph osd set noout
# 停 OSD
ceph orch stop osd
# 停 MGR
ceph orch stop mgr
# 停 MON
ceph orch stop mon
# 各节点重启
reboot


恢复启动顺序
等 MON 起来
ceph -s
# 启动 OSD
ceph orch start osd
# 取消 noout
ceph osd unset noout

ceph orch pause
ceph orch resume
```

Ceph 对磁盘的要求
至少 ≥ 5GB
不能有分区 / 文件系统
不能被挂载
不能是系统盘
