## 3. Zabbix Server配置

### 3.1 部署配置Zabbix Server组件

{{< cmd role="prod-zbx-server-01" title="zbx-server-01 部署配置Zabbix组件" >}}
# 配置apt阿里源
bash <(curl -sSL https://linuxmirrors.cn/main.sh) \
  --source mirrors.aliyun.com \
  --protocol https            \
  --use-intranet-source false \
  --backup true               \
  --upgrade-software false    \
  --clean-cache false         \
  --ignore-backup-tips        \
  --pure-mode

# 下载 key
wget https://mirrors.aliyun.com/zabbix/zabbix-official-repo.key

# 转换
gpg --dearmor -o /usr/share/keyrings/zabbix.gpg zabbix-official-repo.key

# 写源
cat <<EOF > /etc/apt/sources.list.d/zabbix.list
deb [signed-by=/usr/share/keyrings/zabbix.gpg] https://mirrors.aliyun.com/zabbix/zabbix/6.4/ubuntu noble main
EOF

# 更新
apt update

# 安装 Zabbix Server、Zabbix Web、Zabbix Agent 三个组件
sudo apt install -y zabbix-server-mysql zabbix-frontend-php zabbix-agent
sudo apt install -y php-fpm php-mysql php-gd php-xml php-bcmath php-mbstring php-ldap zabbix-nginx-conf
{{< /cmd >}}

### 3.2 部署配置数据库

{{< cmd role="prod-zbx-server-01" title="prod-zbx-server-01 部署配置Zabbix MySQL数据库" >}}
# 安装Zabbix MySQL数据库
sudo apt update

sudo apt install -y mysql-server

sudo mysql -e "ALTER USER 'root'@'localhost' IDENTIFIED WITH mysql_native_password BY 'StrongPassword'; FLUSH PRIVILEGES;"

cat <<EOF | sudo tee  /opt/zabbix.sql
-- 创建Zabbix数据库
CREATE DATABASE zabbix CHARACTER SET utf8mb4 COLLATE utf8mb4_bin;
-- 创建数据库用户zabbix
CREATE USER 'zabbix'@'%' IDENTIFIED BY 'ZabbixStrongPassword';
-- 授权数据库用户zabbix
GRANT ALL PRIVILEGES ON zabbix.* TO 'zabbix'@'%';

FLUSH PRIVILEGES;
EOF

mysql -uroot -p'StrongPassword' -e "source /opt/zabbix.sql;"

# 安装 SQL 脚本包
apt install -y zabbix-sql-scripts

# 开启 trust
# Zabbix 初始化 SQL 包含stored functions、triggers，创建 function 需要zabbix用户有 SUPER 权限
echo "log_bin_trust_function_creators = 1" >> /etc/mysql/mysql.conf.d/mysqld.cnf
# 重启mysql，配置生效
systemctl restart mysql

# 导入Zabbix建表语句
zcat /usr/share/zabbix-sql-scripts/mysql/server.sql.gz | mysql -uzabbix -p'ZabbixStrongPassword' zabbix

# 修改 Zabbix Server 配置文件
cat <<'EOF' >> /etc/zabbix/zabbix_server.conf
DBHost=prod-zbx-server-01
DBPassword=ZabbixStrongPassword
EOF

# 修改 Zabbix agent 配置文件
sed -i 's/^Server=.*/Server=prod-zbx-server-01/' /etc/zabbix/zabbix_agentd.conf
# 关闭主动模式（Active Check）
sed -i 's/^ServerActive=/#ServerActive=/' /etc/zabbix/zabbix_agentd.conf
# 配置Agent 使用系统 hostname
sed -i 's/^Hostname=/#Hostname=/'  /etc/zabbix/zabbix_agentd.conf

sed -i 's|^;date.timezone =.*|date.timezone = Asia/Shanghai|' /etc/php/8.3/apache2/php.ini

# 修改nginx配置文件
cat /etc/nginx/conf.d/zabbix.conf
server {
    listen          80;                  # 或你想用的端口
    server_name     192.168.101.40;       # 改成你的实际情况

    root            /usr/share/zabbix;   # 重要！Zabbix 7.x 前端路径通常是这里（旧版可能无 /ui）

    index           index.php;

    location / {
        try_files $uri $uri/ =404;
    }

    location ~ \.php$ {
        fastcgi_pass   unix:/var/run/php/php8.3-fpm.sock;   # 确认 socket 路径正确（php -i | grep sock）
        fastcgi_index  index.php;
        fastcgi_param  SCRIPT_FILENAME  $document_root$fastcgi_script_name;
        include        /etc/nginx/fastcgi_params;   # 或 snippets/fastcgi-php.conf
        fastcgi_param  PHP_VALUE "date.timezone=Asia/Shanghai";  # 可加时区
    }

    location ~ /\.ht {
        deny all;
    }
}

systemctl enable --now zabbix-server zabbix-agent nginx php8.3-fpm
systemctl status zabbix-server zabbix-agent nginx php8.3-fpm
{{< /cmd >}}
