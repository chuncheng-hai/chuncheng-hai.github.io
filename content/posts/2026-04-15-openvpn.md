## 服务端配置

EC2 Security Group 已放行 UDP 1194

```bash
# 安装openvpn 等工具，期间涉及到iptables的配置选择，两个全部选择yes回车 ,iptables-save持久化/etc/iptables/rules.v4
apt update && apt install -y \
  openvpn \
  easy-rsa \
  iptables \
  iptables-persistent

# 验证是否安装成功
/usr/share/easy-rsa/easyrsa --version
openvpn --version
iptables --version

# 准备 Easy-RSA 环境
make-cadir /etc/openvpn/easy-rsa
cd /etc/openvpn/easy-rsa

# 初始化 PKI
# 配置ECDSA（Elliptic Curve Digital Signature Algorithm，椭圆曲线数字签名算法）
export EASYRSA_ALGO=ec
export EASYRSA_CURVE=prime256v1

# 指定生成目录
export EASYRSA_PKI=/etc/openvpn/easy-rsa/pki
export EASYRSA_BATCH=1
./easyrsa init-pki
./easyrsa build-ca nopass

# 验证ca.crt ECDSA是否生效
openssl x509 -in /etc/openvpn/easy-rsa/pki/ca.crt -text -noout | grep "Public Key Algorithm"

# 生成服务端证书
./easyrsa gen-req server nopass
./easyrsa sign-req server server
# 验证server.crt  ECDSA是否生效
openssl x509 -in /etc/openvpn/easy-rsa/pki/issued/server.crt -text -noout | grep "Public Key Algorithm"

# 生成客户端证书
./easyrsa gen-req client1 nopass
./easyrsa sign-req client client1
openssl x509 -in /etc/openvpn/easy-rsa/pki/issued/client1.crt -text -noout | grep "Public Key Algorithm"

# 生成 tls-crypt-v2 server key
openvpn --genkey tls-crypt-v2-server /etc/openvpn/tls-crypt-v2-server.key

# 为 client 生成 key，将输出内容保存到客户端的配置文件 <tls-crypt-v2>(粘贴完整内容)</tls-crypt-v2>
openvpn --tls-crypt-v2 /etc/openvpn/tls-crypt-v2-server.key --genkey tls-crypt-v2-client

cat <<EOF | sudo tee  /etc/openvpn/server/server.conf
port 1194
proto udp
dev tun

# === cert ===
ca /etc/openvpn/easy-rsa/pki/ca.crt
cert /etc/openvpn/easy-rsa/pki/issued/server.crt
key /etc/openvpn/easy-rsa/pki/private/server.key

# === tls-crypt-v2 ===
tls-crypt-v2 /etc/openvpn/tls-crypt-v2-server.key
dh none

# === network ===
server 10.8.0.0 255.255.255.0
topology subnet

push "route 172.31.0.0 255.255.0.0"

# === security ===
tls-version-min 1.2
tls-cipher TLS-ECDHE-ECDSA-WITH-AES-256-GCM-SHA384

data-ciphers AES-128-GCM:AES-256-GCM
data-ciphers-fallback AES-128-GCM

keepalive 10 60
persist-key
persist-tun

user nobody
group nogroup

verb 3
EOF

# 开启ip转发
echo "net.ipv4.ip_forward = 1" >> /etc/sysctl.conf 
sysctl -p

# 查看dev后的网卡名称
ip route | grep default | grep dev

iptables -t nat -A POSTROUTING -s 10.8.0.0/24 -o enX0 -j MASQUERADE
netfilter-persistent save

# 关闭EC2 关闭源/目的检查

chmod 600 /etc/openvpn/pki/private/server.key
chmod 644 /etc/openvpn/pki/issued/server.crt
chmod 644 /etc/openvpn/pki/ca.crt

systemctl enable --now openvpn-server@server
systemctl status openvpn-server@server

ss -lunpt | grep 1194
```
