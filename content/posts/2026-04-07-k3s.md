```bash
curl -o install.sh https://get.k3s.io

apt install -y zstd aria2

aria2c -x 16 -s 16 -k 1M https://github.com/k3s-io/k3s/releases/download/v1.35.3%2Bk3s1/k3s
chmod +x k3s
mv k3s /usr/local/bin/
k3s --version

aria2c -x 16 -s 16 -k 1M https://github.com/k3s-io/k3s/releases/download/v1.35.3%2Bk3s1/k3s-airgap-images-amd64.tar.zst

sudo mkdir -p /var/lib/rancher/k3s/agent/images/

unzstd k3s-airgap-images-amd64.tar.zst

mv k3s-airgap-images-amd64.tar /var/lib/rancher/k3s/agent/images/

chmod +x install.sh
# 关键环境变量：跳过在线下载
export INSTALL_K3S_SKIP_DOWNLOAD=true

# 推荐加上 --write-kubeconfig-mode 644，方便后续使用 kubectl
INSTALL_K3S_SKIP_DOWNLOAD=true ./install.sh server \
  --write-kubeconfig-mode 644 \
  --cluster-init   # 单节点可加此参数
systemctl status k3s
kubectl get node
kubectl get pods -A

# 在 Server 节点获取 token
cat /var/lib/rancher/k3s/server/node-token
export INSTALL_K3S_SKIP_DOWNLOAD=true
# 配置高可用，添加更多 Server 节点
INSTALL_K3S_SKIP_DOWNLOAD=true ./install.sh server \
  --server https://<第一个-server-IP>:6443 \
  --token <node-token>

# 添加 Worker 节点（Agent）
INSTALL_K3S_SKIP_DOWNLOAD=true ./install.sh agent \
  --server https://<你的-server-IP>:6443 \
  --token <上面获取的token>



```
